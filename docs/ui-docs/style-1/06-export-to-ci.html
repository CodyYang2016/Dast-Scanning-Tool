The UI doesn't schedule anything. It writes a workflow for the app's own repository that runs the same commands you ran here, so CI is the system of record for unattended scans.

.github/workflows/dast-dvwa.yml

validated
name: dast-dvwa
on:
  schedule: [{ cron: "0 6 * * 1" }]   # Mondays 06:00 UTC
  workflow_dispatch:
permissions: { contents: read, security-events: write }
jobs:
  scan:
    runs-on: ubuntu-latest
    timeout-minutes: 45
    steps:
      - uses: actions/checkout@v4
      - name: ZAP key for this run
        run: echo "ZAP_API_KEY=$(openssl rand -hex 24)" >> "$GITHUB_ENV"
      - name: start ZAP
        run: docker compose up -d zap
      - name: author, scan
        env: { DVWA_USER: ${{ secrets.DVWA_USER }}, DVWA_PASS: ${{ secrets.DVWA_PASS }} }
        run: |
          python -m dast author dvwa --no-llm
          python -m dast scan   dvwa
      - name: report and publish
        run: |
          python -m dast report dvwa --upload \
            --ref "$DEPLOYED_REF" --commit "$DEPLOYED_SHA"   # from your deploy job
